on_trait_gained = {
	on_actions = {
		on_trait_gained_add_immune_to_disease
		immortality21_remove_negative_traits
	}
}

on_trait_gained_add_immune_to_disease = {
	trigger = {
		scope:trait = {
			has_trait_flag = immune_to_disease
		}
	}
	effect = {
		add_character_flag = immune_to_disease
	}
}

immortality21_remove_negative_traits = {
	trigger = {
		has_trait_with_flag = immune_to_disease
		scope:trait = {
			OR = {
				AND = {
					has_trait_category = health
					NOT = {
						this = trait:pregnant
					}
				}
				negative_physical_trait_trigger = yes
			}
		}
	}
	effect = {
		remove_trait = scope:trait
	}
}

set_immortal = {
	effect = {
		remove_trait = NonImmortality21
		ad_immortal = yes
	}
}

#################### 出生触发
# 出生时触发 root 为出生角色
on_birth_child = {
	on_actions = {
		son_of_immortality
	}
}

son_of_immortality = {
	trigger = {
		OR = {
			has_game_rule = is_yes_all
			has_game_rule = is_yes_fa
		}
		OR = {
			# 获得作用域角色的真正父亲
			real_father ?= {
				has_trait_with_flag = immortality21
			}
			# 存在母亲
			mother ?= {
				has_trait_with_flag = immortality21
			}
		}
	}
	effect = {
		trigger_event = add_son_of_immortality.01
	}
}
