#################### 出生触发
# 出生时触发 root 为出生角色
on_birth_child = {
	on_actions = {
		son_of_immortality
	}
}

son_of_immortality = {
	trigger = {
		OR = {
			has_game_rule = is_yes_all
			has_game_rule = is_yes_fa
		}
		OR = {
			# 获得作用域角色的真正父亲
			real_father ?= {
				has_trait = immortality21
			}
			# 存在母亲
			mother ?= {
				has_trait = immortality21
			}
		}
	}
	effect = {
		trigger_event = add_son_of_immortality.01
	}
}

on_birthday = {
	on_actions = {
		remove_negative_traits
	}
}

remove_negative_traits = {
	trigger = {
		OR = {
			has_trait = son_of_immortality1
			has_trait = son_of_immortality2
			has_trait = immortality21
		}
	}
	effect = {
		random_trait = {
			limit = {
				has_trait_flag = is_healthy_trigger_flag
				root = {
					has_trait = prev
				}
			}
			prev = {
				remove_trait = prev
			}
		}
		random_trait = {
			limit = {
				has_trait_flag = illness
				root = {
					has_trait = prev
				}
			}
			prev = {
				remove_trait = prev
			}
		}

		if = {
			limit = {
				has_bad_physical_attribute_trigger = yes
			}
			remove_bad_physical_traits_effect = yes
		}
	}
}
